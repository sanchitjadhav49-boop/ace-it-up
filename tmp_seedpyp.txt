61:       if (!Array.isArray(q) || q.length !== 3 || typeof q[1] !== 'number' ||
62:           !DIFFICULTIES.includes(q[2])) {
63:         throw new Error(`Paper ${index + 1} (${sec.name}): malformed numerical entry ${JSON.stringify(q)}`);
64:       }
65:     }
66:   }
67:   return true;
68: }

70: async function seedPaper(client, paper) {
71:   // Remove any existing test with this title (cascade deletes sections, questions, options).
72:   await client.query('DELETE FROM tests WHERE title = $1', [paper.title]);
73: 
74:   const testRes = await client.query(
75:     `INSERT INTO tests (title, description, duration_minutes)
76:      VALUES ($1, $2, 180) RETURNING id`,
77:     [paper.title, paper.description]
78:   );
79:   const testId = testRes.rows[0].id;
80: 
81:   let total = 0;
82:   for (const section of paper.sections) {
83:     const secRes = await client.query(
84:       `INSERT INTO sections (test_id, name, position)
85:        VALUES ($1, $2, $3) RETURNING id`,
86:       [testId, section.name, paper.sections.indexOf(section) + 1]
87:     );
88:     const sectionId = secRes.rows[0].id;
89: 
90:     let position = 1;
91:     // 20 MCQs
92:     for (const [body, options, correctIdx, difficulty] of section.mcq) {
93:       const qRes = await client.query(
94:         `INSERT INTO questions (section_id, question_type, body, difficulty, position)
95:          VALUES ($1, 'mcq', $2, $3, $4) RETURNING id`,
96:         [sectionId, body, difficulty, position]
97:       );
98:       const qId = qRes.rows[0].id;
99:       let correctOptionId = null;
100:       for (let i = 0; i < options.length; i++) {
101:         const oRes = await client.query(
102:           `INSERT INTO question_options (question_id, body, position)
103:            VALUES ($1, $2, $3) RETURNING id`,
104:           [qId, options[i], i + 1]
105:         );
106:         if (i === correctIdx) correctOptionId = oRes.rows[0].id;
107:       }
108:       await client.query(
109:         `UPDATE questions SET correct_option_id = $1 WHERE id = $2`,
110:         [correctOptionId, qId]
111:       );
112:       position++;
113:       total++;
114:     }
115:     // 5 numerical
116:     for (const [body, answer, difficulty] of section.num) {
117:       await client.query(
118:         `INSERT INTO questions (section_id, question_type, body, correct_answer, difficulty, position)
119:          VALUES ($1, 'numerical', $2, $3, $4, $5)`,
120:         [sectionId, body, answer, difficulty, position]
121:       );
122:       position++;
123:       total++;
124:     }
125:   }
126: 
127:   // Validate the full structural contract, then publish.
128:   await client.query('SELECT fn_validate_test_for_publish($1)', [testId]);
129:   await client.query('UPDATE tests SET is_published = TRUE WHERE id = $1', [testId]);
130: 
131:   return { testId, total };
132: }

134: async function main() {
135:   const papers = loadPapers();
136:   const client = await pool.connect();
137:   try {
138:     await client.query('BEGIN');
139:     for (let i = 0; i < papers.length; i++) {
140:       validatePaperShape(papers[i], i);
141:     }
142:     for (let i = 0; i < papers.length; i++) {
143:       const { testId, total } = await seedPaper(client, papers[i]);
144:       console.log(`Seeded test #${testId} "${papers[i].title}" with ${total} questions (published).`);
145:     }
146:     await client.query('COMMIT');
147:     console.log(`Done: ${papers.length} papers published.`);
148:   } catch (err) {
149:     await client.query('ROLLBACK').catch(() => {});
150:     console.error('Seed failed:', err.message);
151:     process.exitCode = 1;
152:   } finally {
153:     client.release();
154:     await pool.end();
155:   }
156: }
