14:   try {
15:     const { email, full_name, password } = req.body;
16:     if (!email || !full_name || !password) {
17:       throw new ApiError(400, 'email, full_name and password are required');
18:     }
19: 
20:     const password_hash = await bcrypt.hash(password, 10);
21: 
22:     const result = await pool.query(
23:       `INSERT INTO users (email, full_name, password_hash)
24:        VALUES ($1, $2, $3)
25:        RETURNING id, email, full_name`,
26:       [email, full_name, password_hash]
27:     );
28: 
29:     res.json({ user: result.rows[0] });
30:   } catch (err) {
31:     if (err.code === '23505') {
32:       // duplicate email
33:       return res.status(400).json({ error: 'An account with this email already exists' });
34:     }
35:     res.status(err.status || 500).json({ error: err.message });
36:   }
37: });

42: app.post('/api/login', async (req, res) => {
43:   try {
44:     const { email, password } = req.body;
45:     if (!email || !password) {
46:       throw new ApiError(400, 'email and password are required');
47:     }
48: 
49:     const result = await pool.query(
50:       `SELECT id, email, full_name, password_hash FROM users WHERE email = $1`,
51:       [email]
52:     );
53:     const user = result.rows[0];
54: 
55:     if (!user || !user.password_hash) {
56:       return res.status(401).json({ error: 'Invalid email or password' });
57:     }
58: 
59:     const matches = await bcrypt.compare(password, user.password_hash);
60:     if (!matches) {
61:       return res.status(401).json({ error: 'Invalid email or password' });
62:     }
63: 
64:     res.json({
65:       user: { id: user.id, email: user.email, full_name: user.full_name }
66:     });
67:   } catch (err) {
68:     res.status(err.status || 500).json({ error: err.message });
69:   }
70: });

85:        JOIN tests t ON t.id = a.test_id
