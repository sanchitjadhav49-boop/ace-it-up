25: function formatDuration(totalSeconds) {
26:   const t = Math.max(0, Math.round(Number(totalSeconds) || 0));
27:   const h = Math.floor(t / 3600);
28:   const m = Math.floor((t % 3600) / 60);
29:   const s = t % 60;
30:   if (h > 0) return `${h}h ${m}m ${s}s`;
31:   if (m > 0) return `${m}m ${s}s`;
32:   return `${s}s`;
